import type { Vec3 } from '../../src/physics/math';
const sub = (a: Vec3, b: Vec3): Vec3 => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const cr = (a: Vec3, b: Vec3): Vec3 => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const dt = (a: Vec3, b: Vec3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
function maxFaceVerts(v: Vec3[]): number {
  let best = 0; const tol = 1e-7; const seen: string[] = [];
  const n = v.length;
  for (let i = 0; i < n; i++) for (let j = i + 1; j < n; j++) for (let k = j + 1; k < n; k++) {
    let nn = cr(sub(v[j]!, v[i]!), sub(v[k]!, v[i]!)); const l = Math.hypot(...nn); if (l < 1e-9) continue;
    nn = nn.map((x) => x / l) as Vec3; const d = dt(nn, v[i]!);
    let a = 0, b = 0, on = 0;
    for (const p of v) { const s = dt(nn, p) - d; if (s > tol) a++; else if (s < -tol) b++; else on++; }
    if (a && b) continue; best = Math.max(best, on);
  }
  return best;
}
